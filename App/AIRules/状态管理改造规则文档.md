# 状态管理改造规则文档

## 📋 概述

本文档总结了状态管理文件的标准化改造规则，基于 `loginState.ts` 的优化模式，适用于所有类似的状态管理文件改造。

## 🎯 改造目标

- ✅ **类型安全**：消除运行时类型错误
- ✅ **可维护性**：清晰的代码结构和文档
- ✅ **健壮性**：完善的错误处理机制
- ✅ **一致性**：统一的编码规范和模式
- ✅ **可扩展性**：便于后续功能扩展

### 3. 逻辑错误修复规则

#### 3.1 修复多重 if 语句
```typescript
// ❌ 错误写法
if (data === 0) {
  status.value = 'STATUS_ONE';
}
if (data === 1) {
  status.value = 'STATUS_TWO';
}
if (data === 2) {
  status.value = 'STATUS_THREE';
}

// ✅ 正确写法
if (data === 0) {
  status.value = 'STATUS_ONE';
} else if (data === 1) {
  status.value = 'STATUS_TWO';
} else if (data === 2) {
  status.value = 'STATUS_THREE';
}
```

#### 3.2 修复空语句块
```typescript
// ❌ 错误写法
if (data === 4) { }
status.value = 'ENDED';

// ✅ 正确写法
if (data === 4) {
  status.value = 'ENDED';
}
```

#### 3.3 修复字符串错误
```typescript
// ❌ 错误写法（注意前面的空格）
status.value = ' NOT_STARTED';

// ✅ 正确写法
status.value = 'NOT_STARTED';
// 或者更好的方式，使用枚举

```

### 4. 错误处理强化规则

#### 4.1 安全JSON解析工具函数
```typescript
/**
 * 安全的JSON解析函数
 * @param jsonString JSON字符串
 * @param defaultValue 解析失败时的默认值
 */
function safeJsonParse<T>(jsonString: string, defaultValue: T): T {
  try {
    return JSON.parse(jsonString);
  } catch (error) {
    console.error("JSON parse error:", error);
    return defaultValue;
  }
}
```

#### 4.2 状态码映射工具函数
```typescript
/**
 * 将状态码转换为状态字符串
 * @param statusCode 状态码
 */
function mapStatusCodeTo[ModuleName]Status(statusCode: number): [ModuleName]StatusType | null {
  const mappedStatus = [MODULE_NAME]_STATUS_MAP[statusCode as [ModuleName]StatusCodeType];
  if (!mappedStatus) {
    console.warn(`Unknown [module] status code: ${statusCode}`);
    return null;
  }
  return mappedStatus;
}
```

## 📝 示例对比

### 改造前
```typescript
// 问题：硬编码、缺乏类型、逻辑错误
const status = ref("IDLE");

function handleStatus(res: string) {
  const data = JSON.parse(res);
  if (data === 0) {
    status.value = 'IDLE'
  }
  if (data === 1) {
    status.value = 'ACTIVE'
  }
}
```

### 改造后
```typescript
// 解决方案：枚举、类型安全、错误处理
export const StatusCode = {
  IDLE: 0,
  ACTIVE: 1,
} as const;

export const Status = {
  IDLE: 'IDLE',
  ACTIVE: 'ACTIVE',
} as const;

export type StatusType = typeof Status[keyof typeof Status];

const status = ref<StatusType>(Status.IDLE);

function handleStatus(res: string): void {
  const statusCode = safeJsonParse<number>(res, -1);
  const mappedStatus = mapStatusCodeToStatus(statusCode);
  
  if (mappedStatus) {
    status.value = mappedStatus;
  } else {
    console.error(`Invalid status code: ${statusCode}`);
  }
}
```
